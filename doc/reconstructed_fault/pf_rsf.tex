\section{Basic Equations}\label{sect:basic equations}

\subsection{Model kinematics}\label{sect:kinematics}

Consider a body $\mathcal B\in\mathbb R^2$ with a crack surface $\mathcal F\subset\mathcal B$ that is described as the set of points belonging to a 1D curve. We assume that $\mathcal F$ is smooth enough, which means that for any $\bm x\in\mathcal F$, there is a neighborhood $\mathcal C_{\bm x}\in\mathcal B$ such that any $\bm x'\in\mathcal C_{\bm x}$ has a unique Euclidean projection on $\mathcal F$. Let $\mathcal C_{\mathcal F} = \bigcup_{\bm x\in\mathcal F}\mathcal C_{\bm x}$. For any $\bm x_{\mathcal F}\in\mathcal F$, define $\mathcal T_{\bm x_\mathcal F}\subset\mathcal C_{\mathcal F}$ as the 1D curve composed by all the points $\bm x\in\mathcal C_{\mathcal F}$ that map to a particular crack coordinate $\bm x_\mathcal F$, and define a 1D coordinate system $\{\zeta\}_{\bm x_\mathcal F}$ along $\mathcal T_{\bm x_\mathcal F}$ with the origin at $\bm x_\mathcal F$. The positive direction of $\{\zeta\}_{\bm x_\mathcal F}$ is the normal direction at $\bm x_\mathcal F$, denoted by $\bm e^n_{\bm x_\mathcal F}$.

Next, consider a velocity field ${\bm u}(\bm x)\in\left(H^2(\mathcal B\backslash\mathcal F)\right)^2$. For two points $\bm x^\pm\in\mathcal T_{\bm x_\mathcal F}$ opposite coordinates $\pm\zeta$, define the normal velocities ${\bm u}_{\pm\zeta}^n = ({\bm u}_{\pm\zeta}\cdot\bm e^n_{\bm x_\mathcal F})\bm e^n_{\bm x_\mathcal F}$ and tangential velocities ${\bm u}_{\pm\zeta}^t = {\bm u}_{\pm\zeta} - {\bm u}_{\pm\zeta}^n$. In addition, define the \textit{slip rate} $V$ and \textit{slip direction} $\bm e^s$ at $\bm x_\mathcal F$ by
\begin{equation}
  V_{\bm x_\mathcal F} = \lim_{\zeta\to 0^+}\left|{\bm u}_{+\zeta}^t - {\bm u}_{-\zeta}^t\right|\quad\text{and}\quad
  \bm e^s_{\bm x_\mathcal F} = \lim_{\zeta\to 0^+}\left({\bm u}_{+\zeta}^t - {\bm u}_{-\zeta}^t\right)/V_{\bm x_\mathcal F},
\end{equation}
respectively. Assuming that normal velocity is continuous across $\mathcal F$, that is, $\displaystyle\lim_{\zeta\to 0+}({\bm u}_{+\zeta}^n - {\bm u}_{-\zeta}^n) = \bm 0$, then we have
\begin{equation}\label{eq:V and s}
  V_{\bm x_\mathcal F} = \lim_{\zeta\to 0^+}\left|{\bm u}_{+\zeta} - {\bm u}_{-\zeta}\right|\quad\text{and}\quad
  \bm e^s_{\bm x_\mathcal F} = \lim_{\zeta\to 0^+}\left({\bm u}_{+\zeta} - {\bm u}_{-\zeta}\right)/V_{\bm x_\mathcal F}.
\end{equation}

Now consider the velocity gradient $\bm L = \nabla{\bm u}$. For an arbitrary $\bm x\in\mathcal T_{\bm x_\mathcal F}$ with coordinate $\zeta$, we define a ``crack velocity gradient'' by
\begin{equation}
  \bm L_\zeta^c = \delta(\zeta)V_{\bm x_\mathcal F}\bm e^s_{\bm x_\mathcal F}\otimes\bm e^n_{\bm x_\mathcal F},
\end{equation}
where $\delta(\cdot)$ is the Dirac delta function. The velocity gradient related to continuous (bulk) deformation is indicated by $\bm L^b = \bm L - \bm L^c$. Then it is clear that
\[
  \lim_{\zeta\to 0^+}({\bm u}_{+\zeta} - {\bm u}_{-\zeta}) = \lim_{\zeta\to 0^+}\int_{-\zeta}^{+\zeta}\bm L_{\zeta'}\cdot\bm e^n_{\bm x_\mathcal F}\mathrm d\zeta' = \lim_{\zeta\to 0^+}\int_{-\zeta}^{+\zeta}\delta(\zeta')V_{\bm x_\mathcal F}\bm e^s_{\bm x_\mathcal F}\mathrm d\zeta' = V_{\bm x_\mathcal F}\bm e^s_{\bm x_\mathcal F},
\]
which is consistent with the definitions of $V$ and $\bm e^s$ in \eqref{eq:V and s}.

Following the decomposition of velocity gradient, the deviatoric component of strain rate can be decomposed by ${\bm\varepsilon} = {\bm\varepsilon}^b + {\bm\varepsilon}^c$, where ${\bm\varepsilon}^b = {\rm dev}({\rm sym}(\bm L^b))$ and ${\bm\varepsilon}^c = {\rm dev}({\rm sym}(\bm L^c))$. For convenience, we define the second-order symmetric tensor
\begin{equation}
  \bm{\mathcal S}_{\bm x_\mathcal F} = \frac{\bm e^n_{\bm x_\mathcal F}\otimes\bm e^s_{\bm x_\mathcal F} + \bm e^s_{\bm x_\mathcal F}\otimes\bm e^n_{\bm x_\mathcal F}}{2},
\end{equation}
then the ``crack strain rate'' has the simple form 
\begin{equation}\label{eq:crack strain raw}
  \bm\varepsilon_\zeta^c = \delta(\zeta)V_{\bm x_\mathcal F}\bm{\mathcal S}_{\bm x_\mathcal F}.
\end{equation}

In phase-field damage (PFD) models, the material is supposed to have two phases: an ``intact'' phase and a ``damaged'' phase, the latter being identified by a smooth phase-field $\phi(\bm x)$. Hence, we can regularize the crack strain rate by
\begin{equation}
  \bm\varepsilon^c\approx\chi(\phi)V_{\bm x_\mathcal F}\bm{\mathcal S}_{\bm x_\mathcal F}\quad\text{with}\quad\int_{-\infty}^{+\infty}\chi(\phi)\mathrm d\zeta = 1.
\end{equation}

In the following, we omit the subscript $\bm x_\mathcal F$ for convenience, only keeping in mind that $V$, $\bm e^n$ and $\bm e^s$ are extended from the crack surface.


\subsection{Model rheology}\label{sect:rheology}

\begin{figure}
  \centering
  \includegraphics[width=.7\linewidth]{fig/rheology.png}
  \caption{An illustration of the viscoelastic damage model adopted in this study.}
  \label{fig:rheology}
\end{figure}

We propose a viscoelastic damage model in analogy to the standard viscoelastic plastic model widely adopted in geodynamic modelling, as plotted in \cref{fig:rheology}. The deviatoric component of bulk strain is governed by a Maxwell-type viscoelastic model, namely,
\begin{equation}\label{eq:maxwell}
  \bm\varepsilon^b = \bm\varepsilon^v + \bm\varepsilon^e\quad\text{with}\quad
  \bm\varepsilon^v = \frac{\bm\tau}{2\eta}\quad\text{and}\quad\dot{\bm\varepsilon^e} = \frac{\dot{\bm\tau}}{2G},
\end{equation}
where $\eta$ and $G$ are the viscosity and the elastic shear modulus, respectively, which are assumed to be constant in time. Here, we have neglected the rotation terms in the objective stress rate. Furthermore, we assume that both $\bm\varepsilon^b$ and $\bm\varepsilon^c$ are constant in time within one time step, say the $k$-th time step $t\in[t_{k-1}, t_k]$. Then \eqref{eq:maxwell} can be regarded as an ODE of $\bm\tau$:
\begin{equation}\label{eq:maxwell ode}
  \frac{\bm\tau}{2\eta} + \frac{\dot{\bm\tau}}{2G} = \bm\varepsilon^b,\qquad t\in[t_{k-1},t_k].
\end{equation}
The solution of \cref{eq:maxwell ode} is
\begin{equation}\label{eq:tau}
  \bm\tau = 2\eta(1-\beta)\bm\varepsilon^b + \beta\bm\tau_{k-1}\quad\text{with}\quad\beta = \mathrm e^{-\frac{t-t_{k-1}}{\lambda}},
\end{equation}
where $\lambda = \eta/G$ is the viscoelastic relaxation time.

The crack deformation, characterized by $\bm\varepsilon^c$, is correlated to three mechanisms:
\begin{itemize}
  \item The rate- and state-dependent friction (RSF), which is commonly given by
    \begin{equation}\label{eq:rsf}
      \mu = \mu_0 + a\ln\left(\frac{V}{V_0}\right) + b\ln\left(\frac{V_0\Theta}{D_c}\right).
    \end{equation}
    It can be seen from \eqref{eq:rsf} that the friction coefficient is the linear combination of a reference friction coefficient $f_0$, a logarithmic contribution of the dimensionless slip rate $V /V_0$ , and another logarithmic contribution of a dimensionless quantity $\Theta V_0 / D_c$ representing the evolution state. The slip state is assumed to follow the aging law
    \begin{equation}\label{eq:aging}
      \dot\Theta = 1 - \frac{V\Theta}{D_c}.
    \end{equation}
    \cref{eq:aging} indicates that the friction is rate-weakening if $a - b < 0$, and rate-strengthening if $a - b > 0$.
  \item The Maxwell body representing the work done by the cohesive damage process, which is introduced to prevent a sudden drop of stress. It can be proved that the relaxation time of the Maxwell body must be identical to that of the bulk deformation (see \cref{sect:crack deformation} for details).
  \item The radiation damping, $\eta^{\rm dpr}V$, which is introduced to approximate the inertial effects under the quasi-dynamic settings. Here, the damping viscosity $\eta^{\rm dpr}$ is calculated by $G / 2c_s$, where $c_s$ is the shear wave speed.
\end{itemize}

In summary, the constitutive relationship for the crack deformation is given by
\begin{equation}\label{eq:crack deformation raw}
  (\bm\tau - \bm\tau^{\rm coh}) : \bm{\mathcal S} - \mu\sigma_n - \eta^{\rm dpr}V \leq 0,
\end{equation}
where $\sigma_n = -\bm\sigma : \bm e^n\otimes\bm e^n$ denotes the normal stress on the fault surface. In practice, we adopt a regularized form of the RSF law
\begin{equation}
  \mu = a\sinh^{-1}\left[\frac{V}{2V_0}\exp\left(\frac{\mu_0 + b\ln(\theta V_0/D_c)}{a}\right)\right],
\end{equation}
which makes $\mu\rightarrow 0$ as $V\rightarrow 0$. Thus, without loss of physical plausibility, we can replace the inequality \eqref{eq:crack deformation raw} with an equation
\begin{equation}\label{eq:crack deformation}
  (\bm\tau - \bm\tau^{\rm coh}) : \bm{\mathcal S} - \mu\sigma_n - \eta^{\rm dpr}V = 0.
\end{equation}


\subsection{Incremental variational formulation}

Consider the $k$-th time step that starts from $t_{k-1}$. Assume that $\bm e^n$ and $\bm e^s$ are determined beforehand, and ignore the radiation damping at the nucleation stage. Let the independent variables be $\{\bm u, p, \upsilon, \phi_k\}$, where $p = -{\rm trace}(\bm\sigma)/3$ is the dynamic pressure, and $\upsilon = \chi V$ represents the magnitude of crack strain rate. For an arbitrary subdomain $\mathcal D\subset\mathcal B$ and an arbitrary end time $t_k > t_{k-1}$, the one-step increment functional has the form
\begin{equation}\label{eq:J raw}
  \begin{aligned}
    \Delta\mathcal J_k(\mathcal D)
    =&\underbrace{\int_{t_{k-1}}^{t_k}\int_{\mathcal D}\mathcal E\left(\nabla\bm u, p, \upsilon\right)\mathrm d\mathcal D\mathrm dt}_{\text{Internal work incr.}} \\
     &-\underbrace{\int_{t_{k-1}}^{t_k}\left(\int_{\mathcal D}\bm b\cdot\bm u\mathrm d\mathcal D + \int_{\mathcal\partial P}\bar{\bm t}\cdot\bm u\right)\mathrm dS\mathrm dt}_{\text{External work incr.}} \\
     &+\underbrace{\int_{\mathcal D}\mathcal G_c(\gamma_k - \gamma_{k-1})\mathrm d\mathcal D}_{\text{Surface energy incr.}}.
  \end{aligned}
\end{equation}
In \eqref{eq:J raw}, $\bm b$ and $\bar{\bm t}$ denote the body force and the boundary traction, respectively, and the functional $\mathcal E\left(\nabla\bm u, p, V\right)$ is defined by 
\begin{equation}\label{eq:I}
  \begin{aligned}
    \mathcal E\left(\nabla\bm u, p, \upsilon\right)
    =&\eta(1 - \beta)\Vert\bm\varepsilon - \upsilon\bm{\mathcal S}\Vert^2 + \beta\bm\tau_{k-1} : (\bm\varepsilon - \upsilon\bm{\mathcal S}) \\
    &- p\nabla\cdot\bm u + \frac{1}{2}\tilde\eta(1-\tilde\beta)\upsilon^2 + \upsilon\tilde\beta\bm\tau_{k-1}^{\rm coh} : \bm{\mathcal S} + \mu\sigma_n\upsilon.
  \end{aligned}
\end{equation}
The physical meanings of the terms on the right-hand side of \eqref{eq:I} are interpreted as follows:
\begin{itemize}
  \item $\eta(1 - \beta)\Vert\bm\varepsilon - \upsilon\bm{\mathcal S}\Vert^2 + \beta\bm\tau_{k-1} : (\bm\varepsilon - \upsilon\bm{\mathcal S})$: a one-step stress potential representing the rate of increase in the work done by the deviatoric part of bulk deformation per unit volume. It is easy to verify that the derivative of this potential with respect to $\bm\varepsilon^b_k = \bm\varepsilon - \upsilon\bm{\mathcal S}$ equals $\bm\tau(t)$ over $[t_{k-1},t_k]$ as defined in \eqref{eq:tau}.
  \item $-p\nabla\cdot\bm u$: the rate of increase in the strain energy per unit volume related to the volumetric part of bulk deformation.
  \item $\frac{1}{2}\tilde\eta(1-\tilde\beta)\upsilon^2 + \upsilon\tilde\beta\bm\tau_{k-1}^{\rm coh} : \bm{\mathcal S}$: a one-step stress potential representing the rate of increase in the work done by the cohesive damage process (apart from frictional dissipation) per unit volume, where $\widetilde G$ and $\tilde\eta$ are supposed to be functions of $\phi$, and $\tilde\beta$ is defined by
    \begin{equation}
      \tilde\beta = \mathrm e^{-(t-t_{k-1})/\tilde\lambda}\quad\text{with}\quad\tilde\lambda = \tilde\beta\big/\widetilde G.
    \end{equation}
    It is easy to see that the potential is equivalent to $\tilde\eta\Vert\bm\varepsilon^c\Vert^2 + \tilde\beta\bm\tau_{k-1}^{\rm coh} : \bm\varepsilon^c$, which leads to
    \begin{equation}
      \bm\tau_k^{\rm coh} = 2\tilde\eta(1-\tilde\beta)\bm\varepsilon^c + \tilde\beta\bm\tau_{k-1}^{\rm coh}.
    \end{equation}
  \item $\mu\sigma_n\upsilon$: frictional dissipation rate per unit volume. Here $\mu$ is constant, determined by the initial values of $V$ and $\Theta$:
    \begin{equation}
      \mu = \mu_0 + a\ln\left(\frac{V^{\rm init}}{V_0}\right) + b\ln\left(\frac{V_0\Theta^{\rm init}}{D_c}\right).
    \end{equation}
    $\bm\tau_k^{\rm fric}$, the stress conjugate to $\bm\varepsilon^c$ in the frictional shear mechanism, is given by
    \begin{equation}\label{eq:tau fric}
      \bm\tau_k^{\rm fric} = 2\mu\sigma_n\bm{\mathcal S}.
    \end{equation}
    Thus, the following equality holds:
    \[
      \int_{-\infty}^{+\infty}\bm\tau_k^{\rm fric} : \bm\varepsilon^c\mathrm d\zeta = \mu\sigma_nV.
    \]
\end{itemize}

Finally, in the surface energy increment, the integrand $\mathcal G_c(\gamma_k - \gamma_{k-1})$ represents the increase in the crack surface energy per unit volume. Here, $\mathcal G_c$ is a material property characterizing the surface energy per unit area, known as the \textit{critical energy release rate}, and $\gamma$ denotes the \textit{crack surface density} functional, defined by
\begin{equation}\label{eq:gamma}
  \gamma(\phi,\nabla\phi) = \frac{1}{c_0}\left(\frac{\alpha(\phi)}{\ell} + \ell|\nabla\phi|^2\right)\quad\text{with}\quad
  c_0 = 4\int_0^1\sqrt{\alpha(\phi)}\mathrm d\phi,
\end{equation}
where $\alpha(\phi)$ is known as the \textit{geometric crack function}, $\ell$ is the length parameter that controls the width of the smeared crack zone, and $c_0$ is a normalization factor.


\subsection{Governing equations}

\subsubsection{Governing equations for bulk deformation}

The first-order stability condition requires that $\delta\mathcal J_k(\mathcal D)\geq 0$ for any virtual increase $\{\delta\bm u, \delta p, \upsilon, \phi_k\}$ satisfying $\phi_k - \phi_{k-1}\geq 0$. First, let $\delta p = 0$, $\delta\upsilon = 0$ and $\delta\phi_k = 0$, we have
\[
  \begin{aligned}
    \delta\mathcal J_k 
    =&\int_{t_{k-1}}^{t_k}\int_{\mathcal D}\Big\{\big[2\eta(1-\beta)(\bm\varepsilon - \upsilon\bm{\mathcal S}) + \beta\bm\tau_{k-1}\big] : \delta\bm\varepsilon - p\nabla\cdot\delta\bm u - \bm b\cdot\delta\bm u\Big\}\mathrm d\mathcal D\mathrm dt \\
    &-\int_{t_{k-1}}^{t_k}\int_{\partial\mathcal D}\bar{\bm t}\cdot\delta\bm u\mathrm dS\mathrm dt \\
    =&\int_{t_{k-1}}^{t_k}\int_{\mathcal D}\Big\{-\nabla\cdot\big[2\eta(1-\beta)(\bm\varepsilon - \upsilon\bm{\mathcal S}) + \beta\bm\tau_{k-1} - p\bm I\big] - \bm b\Big\}\cdot\delta\bm u\mathrm d\mathcal D\mathrm dt \\
    &+\int_{t_{k-1}}^{t_k}\int_{\partial\mathcal D}\Big\{\bm e^n\cdot\big[2\eta(1-\beta)(\bm\varepsilon - \upsilon\bm{\mathcal S}) + \beta\bm\tau_{k-1} - p\bm I\big] - \bar{\bm t}\Big\}\cdot\delta\bm u\mathrm dS\mathrm dt \\
    \geq&0.
  \end{aligned}
\]
The above inequality holds for any $\delta\bm u\in\big(H^1(\mathcal B)\big)^2$, $\mathcal D\subset\mathcal B$ and $t_k > t_{k-1}$, which leads to
\begin{subequations}
  \begin{align}
    &\nabla\cdot\big[2\eta(1-\beta)(\bm\varepsilon - \upsilon\bm{\mathcal S}) + \beta\bm\tau_{k-1} - p\bm I\big] + \bm b = \bm 0,\label{eq:equilibrium}\\
    &\bm e^n\cdot\big[2\eta(1-\beta)(\bm\varepsilon - \upsilon\bm{\mathcal S}) + \beta\bm\tau_{k-1} - p\bm I\big] - \bar{\bm t} = \bm 0.\label{eq:cauchy}
  \end{align}
\end{subequations}
It is easy to see that \eqref{eq:equilibrium} and \eqref{eq:cauchy} are the quasi-static equilibrium equation and the definition of Cauchy stress, respectively.


\subsubsection{Governing equations for crack deformation}\label{sect:crack deformation}

Next, we let $\delta\bm u = \bm 0$, $\delta p = 0$ and $\delta\phi_k = 0$. In the non-trivial case $\upsilon > 0$, the first-order stability condition implies that
\[
  \begin{aligned}
    \delta\mathcal J_k(\mathcal D) = \int_{t_{k-1}}^{t_k}\int_{\mathcal D}\Big[
    &-\eta(1-\beta)(2\bm\varepsilon : \bm{\mathcal S} - \upsilon) - \beta\bm\tau_{k-1} : \bm{\mathcal S} \\
    &+ \tilde\eta(1-\tilde\beta)\upsilon + \tilde\beta\bm\tau_{k-1}^{\rm coh} : \bm{\mathcal S} + \mu\sigma_n\Big]\delta\upsilon\mathrm d\mathcal D\mathrm dt \geq 0
  \end{aligned}
\]
holds for any $\delta\upsilon\in L^2(\mathcal B)$, $\mathcal D\subset\mathcal B$ and $t_k > t_{k-1}$. Thus, we have
\begin{equation}\label{eq:dJ deps c}
  -\eta(1-\beta)(2\bm\varepsilon : \bm{\mathcal S} - \upsilon) - \beta\bm\tau_{k-1} : \bm{\mathcal S} + \tilde\eta(1-\tilde\beta)\upsilon + \tilde\beta\bm\tau_{k-1}^{\rm coh} : \bm{\mathcal S} + \mu\sigma_n = 0.
\end{equation}

To determine the relationship between $\bm\tau^{\rm coh}$ and $\upsilon$, we assume that the energetic degradation is characterized by
\begin{equation}\label{eq:stress degradation}
  \bm\tau : \bm{\mathcal S} - \mu\sigma_n = g(\phi)\left(\bm\tau^{\rm tr} : \bm{\mathcal S} - \mu\sigma_n\right)\quad\text{with}\quad\bm\tau^{\rm tr} = 2\eta(1-\beta)\bm\varepsilon + \beta\bm\tau_{k-1}^{\rm tr}.
\end{equation}
In \eqref{eq:stress degradation}, $g(\phi)$ is the energetic degradation function in PFD models, which satisfies the following conditions:
\begin{itemize}
  \item $g(0) = 1$ and $g(1) = 0$;
  \item $g'(\phi) < 0$ for $\phi\in[0,1)$, i.e., $g(\phi)$ is a monotonically decreasing function;
  \item $g'(1) = 0$.
\end{itemize}
\eqref{eq:stress degradation} indicates that the strength of the material is weakened as the phase-field increases, and when the material is fully damaged ($\phi = 1$, $g = 0$), the shear traction on the crack surface is taken over by the frictional shear. Assuming that $\mu$ is constant in time during one time step, we have
\[
  \bm\tau_{k-1}^{\rm coh} : \bm{\mathcal S} = \bm\tau_{k-1} : \bm{\mathcal S} - \mu\sigma_n = g_{k-1}(\bm\tau_{k-1}^{\rm tr} : \bm{\mathcal S} - \mu\sigma_n)
  = g_{k-1}\left(\bm\tau_{k-1}^{\rm tr} - \bm\tau_{k-1} + \bm\tau_{k-1}^{\rm coh}\right) : \bm{\mathcal S}
\]
\begin{equation}\label{eq:tau coh k-1}
  \Longrightarrow\quad g_{k-1}\left(\bm\tau_{k-1}^{\rm tr} - \bm\tau_{k-1}\right) : \bm{\mathcal S} = (1-g_{k-1})\bm\tau_{k-1}^{\rm coh} : \bm{\mathcal S}.
\end{equation}
Substitution of \eqref{eq:stress degradation} and \eqref{eq:tau coh k-1} in \eqref{eq:dJ deps c} gives
\begin{equation}\label{eq:v}
  \left[(1-\tilde\beta)\tilde\eta - \frac{(1-\beta)\eta}{h(\phi)}\right]\upsilon + \left(\tilde\beta - \frac{h_{k-1}}{h(\phi)}\beta\right)\bm\tau_{k-1}^{\rm coh} : \bm{\mathcal S} = 0
  \quad\text{with}\quad h(\phi) = \frac{1}{g(\phi)} - 1.
\end{equation}
In \cref{eq:v}, $\phi$, $\beta$ and $\tilde\beta$ are dependent on $t$. Since \cref{eq:v} holds for any $t > t_{k-1}$, we must have
\begin{equation}\label{eq:coh relaxation}
  \tilde\beta = \frac{h_{k-1}}{h(\phi)}\beta\quad\text{and}\quad\tilde\eta = \frac{1 - \beta}{h(\phi) - h_{k-1}\beta}\eta.
\end{equation}
Therefore, the constitutive relation of the cohesive damage process can be reformulated by
\begin{equation}\label{eq:tau coh raw}
  \tau^{\rm coh} = \frac{(1-\beta)\eta}{h(\phi)}\upsilon + \frac{h_{k-1}}{h(\phi)}\beta\tau_{k-1}^{\rm coh}\quad\text{with}\quad\tau^{\rm coh} = \bm\tau^{\rm coh} : \bm{\mathcal S}.
\end{equation}

\cref{eq:tau coh} implies that the local crack compliance is proportional to $h(\phi)$. Thus, it is natural to define
\begin{equation}\label{eq:chi}
  \chi(\phi) = \frac{h(\phi)}{I^h}\qquad\text{with}\qquad I^h = \int_{\mathcal T} h(\phi)\mathrm d\zeta.
\end{equation}
Substituting \eqref{eq:chi} in \eqref{eq:tau coh raw}, we get
\begin{equation}\label{eq:tau coh}
  \tau_k^{\rm coh} = \frac{1}{I^h_{k}}\left[(1-\beta_k)\eta V_k + \frac{h_{k-1}}{h_k}\beta_k\tau_{k-1}^{\rm coh}\right].
\end{equation}


\subsubsection{Governing system for the phase-field}\label{sect:phase-field}

Finally, we let $\delta\bm u = \bm 0$, $\delta p = 0$ and $\delta\upsilon = 0$. In the internal work increment, the terms relevant to the phase-field are (using \eqref{eq:coh relaxation})
\begin{equation}\label{eq:I phi}
  \int_{\mathcal D}\int_{t_{k-1}}^{t_k}\frac{1}{h(\phi)}\left[\frac{1}{2}(1-\beta)\eta\upsilon^2 + h_{k-1}\beta\tau_{k-1}^{\rm coh}\upsilon\right]\mathrm dt\mathrm d\mathcal D.
\end{equation}
In the above expression, both $\phi$ and $\beta$ are dependent on $t$. Since there is no way to prescribe the function $\phi(t)$, the integration cannot be calculated precisely. As an approximation, we let $\phi(t)\approx\phi_k$ and $\beta(t)\approx\beta_k$, and then the variation of the internal work density can be calculated by
\begin{equation}\label{eq:dJ dphi 1}
  \begin{aligned}
    \delta\int_{\mathcal D}\mathcal E(\nabla\bm u,p,\upsilon)\mathrm d\mathcal D
    \approx&\Delta t_k\int_{\mathcal D}\left[\frac{1}{2}(1-\beta_k)\eta\upsilon^2 + \beta_kh_{k-1}\tau_{k-1}^{\rm coh}\upsilon\right]\delta\left(\frac{1}{h(\phi_k)}\right)\mathrm d\mathcal D \\
    =&-\Delta t_k\int_{\mathcal D}\frac{h'_k}{h_k^2}\left[\frac{1}{2}(1-\beta_k)\eta\upsilon^2 + \beta_kh_{k-1}\tau_{k-1}^{\rm coh}\upsilon\right]\delta\phi_k\mathrm d\mathcal D.
  \end{aligned}
\end{equation}
Conventionally, the derivative of internal work with respect to the phase-field is denoted by $g'\mathcal H$, where $\mathcal H$ is called the \textit{crack driving force}. From \eqref{eq:dJ dphi 1} and the definition of $h(\phi)$, we can see that the crack driving force has the form
\begin{equation}\label{eq:H}
  \mathcal H_k = \frac{\Delta t_k}{(1-g_k)^2}\left[\frac{1}{2}(1-\beta_k)\eta\upsilon^2 + \beta_kh_{k-1}\tau_{k-1}^{\rm coh}\upsilon\right].
\end{equation}
On the other hand, the variation of surface energy increment can be reformulated by
\begin{equation}\label{eq:dJ dphi 2}
  \begin{aligned}
    \delta\int_{\mathcal D}\mathcal G_c(\gamma_k - \gamma_{k-1})\mathrm d\mathcal D
    =&\int_{\mathcal D}\mathcal G_c\delta\gamma(\phi_k,\nabla\phi_k)\mathrm d\mathcal D \\
    =&\int_{\mathcal D}\mathcal G_c\left(\frac{\partial\gamma}{\partial\phi_k}\delta\phi_k + \frac{\partial\gamma}{\partial\nabla\phi_k}\cdot\nabla\delta\phi_k\right)\mathrm d\mathcal D \\
    =&\int_{\mathcal D}\mathcal G_c\left(\frac{\partial\gamma}{\partial\phi_k} - \nabla\cdot\frac{\partial\gamma}{\partial\nabla\phi_k}\right)\delta\phi_k\mathrm d\mathcal D + \int_{\partial\mathcal D}\mathcal G_c\bm e^n\cdot\frac{\partial\gamma}{\partial\nabla\phi_k}\delta\phi_k\mathrm dS.
  \end{aligned}
\end{equation}
The last equality above results from Gauss's theorem. Combining \eqref{eq:dJ dphi 1}, \eqref{eq:H} and \eqref{eq:dJ dphi 2}, we obtain
\begin{equation}\label{eq:dJ dphi}
  \delta\mathcal J_k(\mathcal D)\approx\int_{\mathcal D}\left\{g_k'\mathcal H_k + \mathcal G_c\delta_\phi(\gamma_k)\right\}\delta\phi_k\mathrm d\mathcal D 
  + \int_{\partial\mathcal D}\mathcal G_c\bm e^n\cdot\frac{\partial\gamma}{\partial\nabla\phi_k}\delta\phi_k\mathrm dS\geq 0,\quad\forall\delta\phi_k\geq 0,\quad\forall\mathcal D\subset\mathcal B,
\end{equation}
where $\delta_\phi(\cdot)$ represents the functional derivative with respect to $\phi$, defined by
\begin{equation}
  \delta_\phi(\cdot) = \frac{\partial}{\partial\phi} - \nabla\cdot\frac{\partial}{\partial\nabla\phi}.
\end{equation}
Assuming $\bm e^n\cdot\nabla\phi = 0$ on $\partial\mathcal B$, it can be proved that \eqref{eq:dJ dphi} is equivalent to the following conditions (known as the \textit{Karush-Kuhn-Tucker} (KKT) conditions):
\begin{equation}\label{eq:kkt}
  \dot\phi\geq 0,\qquad
  g'_k\mathcal H_k + \mathcal G_c\delta_\phi(\gamma_k)\geq 0,\qquad
  \big(g'_k\mathcal H_k + \mathcal G_c\delta_\phi(\gamma_k)\big)\dot\phi = 0.
\end{equation}
In practice, the KKT conditions \eqref{eq:kkt} are often replaced by a history-field-based formulation
\begin{equation}\label{eq:H+}
  g'_k\mathcal H_k^+ + \mathcal G_c\delta_\phi(\gamma_k) = 0\quad\text{with}\quad\mathcal H_k^+ = \max_{t\in[0,t_k)}\mathcal H(t),
\end{equation}
which is easier to solve.


\subsection{Specification of the degradation function}

It can be seen from \eqref{eq:kkt} that damage occurs only when $-g'_k\mathcal H_k$ grows bigger than $\mathcal G_c\delta_\phi(\gamma_k)$, which implies an internal threshold for the damage process. In geomechanics, this threshold is characterized by \textit{cohesion} $c$, which is defined as
\begin{equation}
  c = \tau^{\rm cr} - \mu\sigma_n,
\end{equation}
where $\tau^{\rm cr}$ is the critical stress at which the damage occurs.

In PFD models, the cohesion can be determined by the derivative of $g(\phi)$ at $\phi=0$. To show this, we need to reformulate the crack driving force in terms of stress. From \eqref{eq:tau coh} we can get
\[
  \upsilon = \frac{h_k\tau_k^{\rm coh} - \beta_kh_{k-1}\tau_{k-1}^{\rm coh}}{(1-\beta_k)\eta}.
\]
Substitution of the above equation in \eqref{eq:H} gives
\begin{equation}\label{eq:H evolve}
  \mathcal H_k = \frac{\Delta t_k}{2(1-\beta_k)\eta}\frac{\left(h_k\tau_k^{\rm coh}\right)^2 - \left(\beta_kh_{k-1}\tau_{k-1}^{\rm coh}\right)^2}{(1-g_k)^2}.
\end{equation}
In the small-step limit, we have $1-\beta_k\approx\Delta t_k/\lambda$, so $(1-\beta_k)\eta\approx G\Delta t_k$. Hence,
\begin{equation}\label{eq:H small-step}
  \mathcal H_k\approx\frac{\left(h_k\tau_k^{\rm coh}\right)^2 - \left(\beta_kh_{k-1}\tau_{k-1}^{\rm coh}\right)^2}{2G(1-g_k)^2}.
\end{equation}
For nucleation from an intact state, we have $\tau_{k-1}^{\rm coh} = 0$, $\tau_k^{\rm coh} = \tau^{\rm cr} - \mu\sigma_n = c$ and $g_k\to 1$, so
\begin{equation}\label{eq:H cr}
  \mathcal H_k^{\rm cr}\approx\frac{c^2}{2G}.
\end{equation}

Now, consider the governing equation \eqref{eq:H+}. At the intact state, the functional derivative of $\gamma(\phi,\nabla\phi)$ is
\begin{equation}\label{eq:delta gamma 0}
  \delta_\phi(\gamma)\Big|_{\phi=0} = \frac{\alpha'(0)}{c_0\ell}.
\end{equation}
Replacing \eqref{eq:H cr} and \eqref{eq:delta gamma 0} in \eqref{eq:H+}, we obtain
\begin{equation}\label{eq:g'(0)}
  g'(0) = -\frac{\mathcal G_c\alpha'(0)}{c_0\ell}\frac{2G}{c^2}.
\end{equation}
If we choose the Lorentz-type degradation function, namely
\begin{equation}\label{eq:g}
  g(\phi) = \frac{(1-\phi)^2}{(1-\phi)^2 + m\phi(1 + p\phi)},
\end{equation}
then \eqref{eq:g'(0)} provides a constraint on the parameter $m$:
\begin{equation}
  m = -g'(0) = \frac{\mathcal G_c\alpha'(0)}{c_0\ell}\frac{2G}{c^2}.
\end{equation}
It should be noted that in the standard AT2 model, i.e. $\alpha(\phi) = \phi^2$, the cohesion does not influence the damage process, since $\alpha'(0) = 0$.


\subsection{Characteristic length scale}
\label{sect:fourier_characteristic_length}

The regularization length controls the transverse distribution of inelastic
slip. A finite distribution can modify the mechanical interaction between
different points along a fault, even when the phase field is frozen and before
rate-and-state feedback is considered. We first derive this mechanical
response. The RSF law subsequently determines which physical length scales
must be represented accurately.

\subsubsection{Assumptions and notation}

Consider an infinite straight fault in a homogeneous incompressible medium.
Let $\xi$ be the slip-parallel coordinate and $\zeta$ the fault-normal
coordinate. The phase profile, geometry, previous stress history, material
coefficients, and time-step size are held fixed. Set
\[
  \eta^{ve} = \eta(1-\beta),\qquad
  \bm{\mathcal S} = \frac12\begin{pmatrix}0&1\\1&0\end{pmatrix},\qquad
  \int_{-\infty}^{+\infty}\chi(\zeta)\,\mathrm d\zeta=1.
\]
The source has no trace. Therefore, on the incompressible velocity space,
the deviatoric symmetric gradient is simply the symmetric gradient.
There is no perturbation of the body force. We prescribe a slip-rate
perturbation to measure the bulk response; the friction equation is not
imposed pointwise during this mechanical probe. The mature-fault case
has no cohesive traction.

Write
\[
  \delta V(\xi) = \widehat V\,\mathrm e^{ik\xi},\qquad
  \delta\upsilon(\xi,\zeta) = \chi(\zeta)\widehat V\,\mathrm e^{ik\xi},\qquad
  \delta\bm\varepsilon^c = \delta\upsilon\,\bm{\mathcal S}.
\]
The physical perturbation is the real part. We take $k\ne0$ and select
the solution that decays away from the band. The normal Fourier transform
and its inverse are
\[
  \overline f(m)=\int_{-\infty}^{+\infty}f(\zeta)\mathrm e^{-im\zeta}
          \,\mathrm d\zeta,\qquad
  f(\zeta)=\frac1{2\pi}\int_{-\infty}^{+\infty}\overline f(m)
          \mathrm e^{im\zeta}\,\mathrm dm.
\]
Here $m$ denotes a normal wavenumber, not the degradation parameter used
elsewhere. The factor $\mathrm e^{ik\xi}$ is suppressed below.
Define
\[
  Q=k^2+m^2,\qquad A_m=\overline\chi(m)\widehat V,\qquad
  U=\overline{\delta u_\xi},\quad
  W=\overline{\delta u_\zeta},\quad P=\overline{\delta p}.
\]
In particular, $A_m$ is the transform of the scalar crack shear rate
$\delta\upsilon$, not a tensor component:
\[
  \overline{\delta\bm\varepsilon^c}=A_m\bm{\mathcal S},
  \qquad
  \overline{\delta\varepsilon^c_{\xi\zeta}}=\frac{A_m}{2}.
\]

\subsubsection{Fourier-space Stokes equations}

With the previous stress history fixed, the constitutive perturbation is
\[
  \delta\bm\tau
     =2\eta^{ve}\bigl(\nabla_s\delta\bm u
                          -\delta\upsilon\bm{\mathcal S}\bigr),
  \qquad
  \delta\bm\sigma=\delta\bm\tau-\delta p\,\bm I.
\]
Its Fourier components are
\begin{subequations}
  \begin{align}
    \overline{\delta\tau_{\xi\xi}}&=2\eta^{ve} ikU,
    \label{eq:dtau_xx freq}\\
    \overline{\delta\tau_{\zeta\zeta}}&=2\eta^{ve} imW,
    \label{eq:dtau_yy freq}\\
    \overline{\delta\tau_{\xi\zeta}}&=\eta^{ve}(imU+ikW-A_m).
    \label{eq:dtau_xy freq}
  \end{align}
\end{subequations}
Substituting \cref{eq:dtau_xx freq,eq:dtau_yy freq,eq:dtau_xy freq} in the 
Stokes equations, we obtain
\begin{subequations}
  \begin{align}
    -\eta^{ve}(2k^2+m^2)U-\eta^{ve} kmW-ikP-i\eta^{ve} mA_m&=0,
    \label{eq:momentum x freq}\\
    -\eta^{ve} kmU-\eta^{ve}(k^2+2m^2)W-imP-i\eta^{ve} kA_m&=0,
    \label{eq:momentum y freq}\\
    kU+mW&=0.\label{eq:mass freq}
  \end{align}
\end{subequations}
The solution of \cref{eq:momentum x freq,eq:momentum y freq,eq:mass freq} is
\begin{equation}\label{eq:fourier_stokes_solution}
  U=-iA_m\frac{m(m^2-k^2)}{(k^2+m^2)^2},\qquad
  W= iA_m\frac{k(m^2-k^2)}{(k^2+m^2)^2},\qquad
  P=-2\eta^{ve} A_m\frac{km}{k^2+m^2}.
\end{equation}
Back substitution of \eqref{eq:fourier_stokes_solution} in 
\eqref{eq:dtau_xy freq} gives
\begin{equation}\label{eq:fourier_shear_response}
  \overline{\delta\tau_{\xi\zeta}} = -4\eta^{ve} A_m\frac{k^2m^2}{(k^2+m^2)^2}.
\end{equation}

\subsubsection{Implication for the band width}

Define the the generalized traction conjugate to the single surface variable
$V(\xi)$ by
\[
  q(\xi) = \int_{-\infty}^{+\infty}\chi(\zeta)\tau_{\xi\zeta}\,\mathrm d\zeta.
\]
For real $\chi$, the inverse Fourier transform gives
\begin{equation}
  \begin{aligned}
    \delta q(\xi)
    &=\frac{\mathrm e^{ik\xi}}{2\pi}
      \int_{-\infty}^{+\infty}\overline\chi(-m)
      \overline{\delta\tau_{\xi\zeta}}(m)\,\mathrm dm\\
    &=-K_{\rm band}(k)\widehat V\,\mathrm e^{ik\xi},
  \end{aligned}
\end{equation}
where
\begin{equation}\label{eq:fourier_band_stiffness}
  K_{\rm band}(k)=\frac{\eta^{ve}}{2\pi}
    \int_{-\infty}^{+\infty}
      \frac{4k^2m^2}{(k^2+m^2)^2}
      |\overline\chi(m)|^2\,\mathrm dm.
\end{equation}
For the sharp distribution $\chi=\delta(\zeta)$,
$\overline\chi=1$ and
\[
  K_{\rm sharp}(k)=\eta^{ve}|k|.
\]
For a localized sufficiently smooth profile, we have
\[
  K_{\rm band}(k)\sim
       \frac{4\eta^{ve}}{k^2}
          \int_{-\infty}^{+\infty}|\chi'(\zeta)|^2\,\mathrm d\zeta
       \qquad (|k|\rightarrow\infty).
\]
For a normalized family
$\chi_\ell(\zeta)=\ell^{-1}f(\zeta/\ell)$, write
$K_{\rm band}/K_{\rm sharp}=R(|k|\ell)$.
The sharp response is recovered for $|k|\ell\rightarrow0$, whereas
short waves at fixed nonzero width can be weakly resisted.

\section{Numerical Implementation}\label{sect:implementation}

\subsection{Overall algorithm}

In \cref{sect:basic equations}, we have assumed that all the points in a profile $\mathcal T_{\bm x_\mathcal F}$ share the same crack-relevant quantities, such as $V$, $\Theta$, $\bm e^n$ and $\tau^{\rm coh}$. To ensure the uniqueness of crack-relevant quantities in each profile, we reconstruct fault curves from the phase-field, triangulate them into 1D meshes, and store the above quantities for each mesh vertex. The algebraic system contains four variables: $\phi$, $\bm u$, $p$, and $V$, the first three living in the 2D bulk mesh, while the last living in the 1D curve mesh.

In practice, we separate the phase-field system and the mechanical system by employing a one-pass staggered algorithm. During the $k$-th time step (time interval $(t_{k-1}, t_k]$), we first solve \cref{eq:H+} to update $\phi_k$, then solve the mechanical system to obtain $\bm u_k$, $p_k$ and $V_k$, and finally update $\mathcal H_k$ using \cref{eq:H evolve} for the next time step. In the rest of the section, we shall elaborate on the numerical strategy for solving the mechanical system, while the solution of the phase-field system will be investigated in later sections.

Discretize the mechanical variables $\bm u$, $p$ and $V$ by
\begin{equation}
  \bm u\approx\sum_{i\in\mathcal I_u}\bm\varphi^u_i\hat u_i,\qquad
  p\approx\sum_{i\in\mathcal I_p}\varphi^p_i\hat p_i,\qquad
  V\approx\sum_{i\in\mathcal I_V}\varphi^V_i\widehat V_i,
\end{equation}
respectively, where $\mathcal I_u$, $\mathcal I_p$ and $\mathcal I_V$ are the corresponding DoF sets. The weak forms of the mechanical system are given by (the subscript $k$ representing the current time step is omitted) 
\begin{align}
  R^u_i
  =&\int_{\mathcal B}\left(\nabla_s\bm\varphi^u_i : \bm\tau - \nabla\cdot\bm\varphi_i^u p - \bm\varphi_i^u\cdot\bm b\right)\mathrm dA - 
    \int_{\partial\mathcal B}\bm\varphi_i^u\cdot\bar{\bm t}\mathrm dL = 0 & &\forall i\in\mathcal I_u,\label{eq:Ru} \\
  R^p_i
  =&-\int_{\mathcal B}\varphi_i^p\nabla\cdot\bm u\mathrm dA = 0 & &\forall i\in\mathcal I_p,\label{eq:Rp}\\
  R^V_i
  =&\int_{\widehat{\mathcal F}}\varphi_i^V\left(\tau - \tau^{\rm coh} - \mu\sigma_n - \eta^{\rm dpr}V\right)\mathrm dL = 0 
   & &\forall i\in\mathcal I_V\label{eq:RV raw},
\end{align}
where $\widehat{\mathcal F}$ denotes the reconstructed fault curve.

In this study, we use the particle-in-cell method for stress advection. Hence, we need to project the integrand in $R^V$ from particles to the fault curves. For simplicity, assume that there is only one fault curve, on which we build a local coordinate system $\{\xi\}$. First, we define $\widetilde{\mathcal C}_{\mathcal F} = \{\bm x~|~\phi(\bm x) > \phi_{\rm min}\}$ as the numerical fault zone, where $\phi_{\rm min} > 0$ is a threshold under which the phase-field is negligible. We assume that $\widetilde{\mathcal C}_{\mathcal F}\subseteq\mathcal C_{\mathcal F}$. For a particle $\bm x_p\in\widetilde{\mathcal C}_{\mathcal F}$, denote its projection on the fault by $\xi_p$. Further, we build a Voronoi tessellation with the particles as centroids. Denote the Voronoi cell around particle $p$ by $\mathcal V_p$. Each $\mathcal V_p$ is assumed to have uniform stress field. We replace \eqref{eq:RV raw} by a volume integral
\begin{equation}\label{eq:RV}
  R^V_i = \sum_{p\in\mathcal P_{\mathcal F}}\int_{\mathcal V_p}\varphi_i^V(\xi_p)\left(\tau_p - \tau_{\xi_p}^{\rm coh} - \mu_{\xi_p}\sigma_{n,p} - \eta^{\rm dpr}V_{\xi_p}\right)\mathrm dA,\quad i\in\mathcal I_V,
\end{equation}
where
\[
  \mathcal P_{\mathcal F} = \bigcup_{\mathcal V_p\cap\widetilde{\mathcal C}_{\mathcal F}\neq\emptyset}\mathcal V_p.
\]
In \eqref{eq:RV}, $\tau_p$ and $\sigma_{n,p}$ are computed with the stress states stored in particle $p$ and fault-related quantities evaluated at $\xi_p$.

We employ the Newton-Raphson method to solve the nonliear \cref{eq:Ru,eq:Rp,eq:RV}. The linearized system in each nonlinear iteration is given by
\begin{equation}\label{eq:linearized system}
  \begin{bmatrix}
    \mathbf J^{uu} & \mathbf J^{up} & \mathbf J^{uV} \\ \mathbf J^{pu} & \mathbf 0 & \mathbf 0 \\ \mathbf J^{Vu} & \mathbf J^{Vp} & \mathbf J^{VV}
  \end{bmatrix}
  \begin{bmatrix}
    \delta\hat{\mathbf u} \\ \delta\hat{\mathbf p} \\ \delta\widehat{\mathbf V}
  \end{bmatrix}
  = -
  \begin{bmatrix}
    \mathbf R^u \\ \mathbf R^p \\ \mathbf R^V
  \end{bmatrix},
\end{equation}
where
\[
  J_{ij}^{\alpha\beta} = \frac{\partial R^\alpha_i}{\partial\widehat\beta_j},\qquad\alpha,\beta = u, p, V.
\]
In \cref{eq:linearized system}, $\mathbf J^{uu}$, $\mathbf J^{up}$ and $\mathbf J^{pu}$ are identical to the Newton Stokes blocks in \texttt{ASPECT}, while $\mathbf J^{uV}$, $\mathbf J^{Vu}$, $\mathbf J^{Vp}$ and $\mathbf J^{VV}$ work solely for the phase-field fault model. We first focus on $\mathbf J^{VV}$. Using \eqref{eq:tau} and \eqref{eq:tau coh}, we have
\begin{equation}\label{eq:J VV}
  \begin{aligned}
    J^{VV}_{ij} 
    =&\sum_{p\in\mathcal P_{\mathcal F}}\int_{\mathcal V_p}\varphi_i^V(\xi_p)
      \left[\frac{\partial\bm\tau_p}{\partial\widehat V_j} : (\bm{\mathcal S}_{\xi_p} + \mu_{\xi_p}\bm{\mathcal N}_{\xi_p})
      - \frac{\partial\tau_{\xi_p}^{\rm coh}}{\partial\widehat V_j}
      - \frac{\partial\mu_{\xi_p}}{\partial\widehat V_j}\sigma_{n,p}
      - \eta^{\rm dpr}\frac{\partial V_{\xi_p}}{\partial\widehat V_j}\right]\mathrm dA \\
    =&-\sum_{p\in\mathcal P_{\mathcal F}}\int_{\mathcal V_p}\varphi_i^V(\xi_p)
      \left[\frac{(1-\beta)\eta}{g_pI^h_{\xi_p}}
      + \frac{\partial\mu_{\xi_p}}{\partial V_{\xi_p}}\sigma_{n,p}
      + \eta^{\rm dpr}\right]\varphi^V_j(\xi_p)\mathrm dA,\quad i,j\in\mathcal I_V.
  \end{aligned}
\end{equation}
Note that the second equality utilizes the identities $\bm{\mathcal S}_{\xi_p} : \bm{\mathcal S}_{\xi_p} = 1/2$, $\bm{\mathcal S}_{\xi_p} : \bm{\mathcal N}_{\xi_p} = 0$ and $\chi_p + 1/I^h_{\xi_p} = 1/(g_pI^h_{\xi_p})$.

\begin{remark}
  In \eqref{eq:J VV}, we have neglected a term 
  \[
    \frac{\partial\mu_{\xi_p}}{\partial\Theta_{\xi_p}}\frac{\mathrm d\Theta_{\xi_p}}{\mathrm dV_{\xi_p}}.
  \]
  This is because we need to invert $\mathbf J^{VV}$ when solving the linearized system (see below). In coseismic periods, this term could be a large negative number near the rate-softening regions, which increases the condition number of $\mathbf J^{VV}$ and makes the inversion unstable. To avoid this, we exclude this term from the Jacibian, and update $\Theta$ according to the new $V$ after each nonlinear iteration. Therefore, the solving algorthm can be regarded as a modified Newton method with lagged slip state.
\end{remark}

Similarly, we can write down the explicit forms of $\mathbf J^{Vu}$ and $\mathbf J^{Vp}$ as
\begin{align}
  J^{Vu}_{ij} =&\sum_{p\in\mathcal P_{\mathcal F}}\int_{\mathcal V_p}\varphi^V_i(\xi_p)\left[2(1-\beta)\eta\nabla_s\bm\varphi^u_j(\bm x_p) : \left(\bm{\mathcal S}_{\xi_p} + \mu_{\xi_p}\bm{\mathcal N}_{\xi_p}\right)\right]\mathrm dA
  & &i\in\mathcal I_V,~j\in\mathcal I_u,\label{eq:delta X}\\
  J^{Vp}_{ij} =&-\sum_{p\in\mathcal P_{\mathcal F}}\int_{\mathcal V_p}\mu_{\xi_p}\varphi^V_i(\xi_p)\varphi^p_j(\bm x_p)\mathrm dA
  & &i\in\mathcal I_V,~j\in\mathcal I_p.\label{eq:delta V}
\end{align}
The last unknown Jacobian block, $J^{uV}_{ij}$, can be derived straightforwardly
\begin{equation}\label{eq:JuV}
  J^{uV}_{ij} = \int_{\mathcal B}2(1-\beta)\eta\chi\left(\nabla_s\bm\varphi_i^u(\bm x) : \bm{\mathcal S}_{\xi_{\bm x}}\right)\varphi^V_j(\xi_{\bm x})\mathrm dA
  \qquad i\in\mathcal I_u,~j\in\mathcal I_V.
\end{equation}

Define
\[
  \widehat{\mathbf X} =
  \begin{bmatrix}
    \hat{\mathbf u} \\ \hat{\mathbf p}
  \end{bmatrix},\quad
  \mathbf R^X =
  \begin{bmatrix}
    \mathbf R^u \\ \mathbf R^p
  \end{bmatrix},\quad
  \mathbf J^{XX} =
  \begin{bmatrix}
    \mathbf J^{uu} & \mathbf J^{up} \\ \mathbf J^{pu} & \mathbf 0
  \end{bmatrix},\quad
  \mathbf J^{XV} =
  \begin{bmatrix}
    \mathbf J^{uV} \\ \mathbf 0
  \end{bmatrix},\quad
  \mathbf J^{VX} =
  \begin{bmatrix}
    \mathbf J^{Vu} \\ \mathbf J^{Vp}
  \end{bmatrix}^T.
\]
Then the linearized system can be rewritten as
\begin{equation}\label{eq:linearized compact}
  \begin{bmatrix}
    \mathbf J^{XX} & \mathbf J^{XV} \\ \mathbf J^{VX} & \mathbf J^{VV}
  \end{bmatrix}
  \begin{bmatrix}
    \delta\widehat{\mathbf X} \\ \delta\widehat{\mathbf V}
  \end{bmatrix}
  = -
  \begin{bmatrix}
    \mathbf R^X \\ \mathbf R^V
  \end{bmatrix}.
\end{equation}
The solution of \cref{eq:linearized compact} has a condensed form
\begin{subequations}
  \begin{align}
    \left[\mathbf J^{XX} - \mathbf J^{XV}\left(\mathbf J^{VV}\right)^{-1}\mathbf J^{VX}\right]\delta\widehat{\mathbf X} 
    =&-\mathbf R^X + \mathbf J^{XV}\left(\mathbf J^{VV}\right)^{-1}\mathbf R^V,\\
    \mathbf J^{VV}\delta\widehat{\mathbf V} =&-\mathbf R^V - \mathbf J^{VX}\delta\widehat{\mathbf X}.
  \end{align}
\end{subequations}
If $\varphi^V_i$ ($i\in\mathcal I_V$) is in $Q_1$ space, then $\mathbf J^{VV}$ is a tridiagonal matrix, which is trivial to inverse. \cref{eq:delta X} is solved with the FGMRES method. To improve the convergence rate of the linear solver, we use a preconditioner
\begin{equation}
  \widetilde{\mathbf P}^{XX} = \mathbf P^{XX} + \mathbf P^{XX}\mathbf J^{XV}\left(\mathbf J^{VV} - \mathbf J^{VX}\mathbf P^{XX}\mathbf J^{XV}\right)^{-1}\mathbf J^{VX}\mathbf P^{XX},
\end{equation}
where $\mathbf P^{XX}$ is the original right-preconditioner of $\mathbf J^{XX}$. It is easy to verify that 
\[
  \left[\mathbf J^{XX} - \mathbf J^{XV}\left(\mathbf J^{VV}\right)^{-1}\mathbf J^{VX}\right]\widetilde{\mathbf P}^{XX} = \mathbf I
\]
if $\mathbf P^{XX}$ is the exact inverse of $\mathbf J^{XX}$.

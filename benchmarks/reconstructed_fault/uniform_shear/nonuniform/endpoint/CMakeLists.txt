cmake_minimum_required(VERSION 3.16)
project(endpoint_moment_geometry_check LANGUAGES CXX)
set(VORO_ROOT "" CACHE PATH "The Voro++ installation used by the ASPECT build")
find_path(VORO_INCLUDE_DIR voro++.hh HINTS "${VORO_ROOT}/include/voro++" REQUIRED)
find_library(VORO_LIBRARY voro++ HINTS "${VORO_ROOT}/lib" REQUIRED)
add_executable(verify_voronoi_areas verify_voronoi_areas.cc)
target_compile_features(verify_voronoi_areas PRIVATE cxx_std_11)
target_include_directories(verify_voronoi_areas PRIVATE "${VORO_INCLUDE_DIR}")
target_link_libraries(verify_voronoi_areas PRIVATE "${VORO_LIBRARY}")
add_executable(verify_plane_cuts verify_plane_cuts.cc)
target_compile_features(verify_plane_cuts PRIVATE cxx_std_11)
target_include_directories(verify_plane_cuts PRIVATE "${VORO_INCLUDE_DIR}")
target_link_libraries(verify_plane_cuts PRIVATE "${VORO_LIBRARY}")

cmake_minimum_required(VERSION 3.13.4)
find_package(Aspect REQUIRED HINTS ${Aspect_DIR})
DEAL_II_INITIALIZE_CACHED_VARIABLES()
project(transported_objective_native)
add_library(transported_objective_native SHARED native_checks.cc)
ASPECT_SETUP_PLUGIN(transported_objective_native)
target_compile_options(transported_objective_native PRIVATE -fno-finite-math-only -ffp-contract=off)
add_library(transported_material_native SHARED material_checks.cc)
ASPECT_SETUP_PLUGIN(transported_material_native)
target_compile_options(transported_material_native PRIVATE -fno-finite-math-only -ffp-contract=off)
add_library(transported_material_energy SHARED material_checks.cc)
ASPECT_SETUP_PLUGIN(transported_material_energy)
target_compile_definitions(transported_material_energy PRIVATE MATERIAL_ENERGY_CHECKS)
target_compile_options(transported_material_energy PRIVATE -fno-finite-math-only -ffp-contract=off)
add_library(transported_isolation_native SHARED material_checks.cc)
ASPECT_SETUP_PLUGIN(transported_isolation_native)
target_compile_definitions(transported_isolation_native PRIVATE MATERIAL_ISOLATION_CHECKS)
target_compile_options(transported_isolation_native PRIVATE -fno-finite-math-only -ffp-contract=off)

add_library(transported_bulk_spin SHARED material_checks.cc)
ASPECT_SETUP_PLUGIN(transported_bulk_spin)
target_compile_definitions(transported_bulk_spin PRIVATE MATERIAL_ENERGY_CHECKS BULK_SPIN_CHECKS)
target_compile_options(transported_bulk_spin PRIVATE -fno-finite-math-only -ffp-contract=off)

cmake_minimum_required(VERSION 3.16)
find_package(Aspect REQUIRED HINTS ${Aspect_DIR})
DEAL_II_INITIALIZE_CACHED_VARIABLES()
project(r7b_checks)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
foreach(name nsinker_bfbt melt_transport_compressible_iterative phase_field_fault_stage_j_restart_create)
 add_library(${name} SHARED ${REPO}/tests/${name}.cc)
 ASPECT_SETUP_PLUGIN(${name})
endforeach()
add_library(cache_lifecycle SHARED cache_lifecycle.cc)
target_include_directories(cache_lifecycle PRIVATE ${REPO}/tests)
ASPECT_SETUP_PLUGIN(cache_lifecycle)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${REPO}/benchmarks/reconstructed_fault/refactoring_r5b2/plugin/singular_current_diagnostic.py ${CMAKE_CURRENT_BINARY_DIR}/singular.cc COMMAND_ERROR_IS_FATAL ANY)
add_library(singular SHARED ${CMAKE_CURRENT_BINARY_DIR}/singular.cc)
target_include_directories(singular PRIVATE ${REPO}/tests)
ASPECT_SETUP_PLUGIN(singular)

add_library(solcx SHARED ${REPO}/benchmarks/solcx/solcx.cc)
ASPECT_SETUP_PLUGIN(solcx)
add_library(ordinary_audit SHARED ordinary_audit.cc)
ASPECT_SETUP_PLUGIN(ordinary_audit)

add_library(empty_owner SHARED cache_lifecycle.cc)
target_compile_definitions(empty_owner PRIVATE R7B_EMPTY_OWNER)
target_include_directories(empty_owner PRIVATE ${REPO}/tests)
ASPECT_SETUP_PLUGIN(empty_owner)

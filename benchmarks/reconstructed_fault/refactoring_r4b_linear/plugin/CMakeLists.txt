cmake_minimum_required(VERSION 3.13.4)
find_package(Aspect REQUIRED HINTS ${Aspect_DIR})
DEAL_II_INITIALIZE_CACHED_VARIABLES()
project(r4b_existing_linear_checks)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
foreach(name phase_field_fault_residual_consistency phase_field_fault_linear_exhaustion phase_field_fault_pressure_gauge phase_field_fault_stage_i)
  add_library(${name} SHARED ${REPO}/tests/${name}.cc)
  ASPECT_SETUP_PLUGIN(${name})
  if(REFERENCE_HEADERS)
    target_include_directories(${name} BEFORE PRIVATE ${REFERENCE_HEADERS})
  endif()
endforeach()
if(NOT REFERENCE_HEADERS)
  add_library(phase_field_fault_stage_i_rollback SHARED ${REPO}/tests/phase_field_fault_stage_i_rollback.cc)
  ASPECT_SETUP_PLUGIN(phase_field_fault_stage_i_rollback)
  add_library(cache_audit SHARED ${REPO}/benchmarks/reconstructed_fault/refactoring_r2b_cache/plugin/cache_audit.cc)
  ASPECT_SETUP_PLUGIN(cache_audit)
  add_subdirectory(${REPO}/benchmarks/reconstructed_fault/bp3/plugin bp3)
endif()

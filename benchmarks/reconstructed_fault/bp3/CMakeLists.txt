cmake_minimum_required(VERSION 3.13.4)
find_package(Aspect REQUIRED HINTS ${Aspect_DIR})
DEAL_II_INITIALIZE_CACHED_VARIABLES()
project(reconstructed_fault_bp3)
option(BP3_BUILD_LEGACY_PLUGIN "Build the historical rotated-chart research plugin" OFF)
if(BP3_BUILD_LEGACY_PLUGIN)
  add_library(bp3 SHARED bp3.cc ../stress_only_interpolator.cc)
  target_compile_definitions(bp3 PRIVATE ASPECT_BP3_RESEARCH_EXECUTION)
  ASPECT_SETUP_PLUGIN(bp3)
endif()

# The server package and development build use the same restored sources.
add_subdirectory(plugin)
set_target_properties(bp3_restore_150x50 PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")

option(BP3_BUILD_TOOLS "Build mesh preparation and environment checks" OFF)
if(BP3_BUILD_TOOLS)
  add_executable(bp3_restore_mesh restore_mesh.cc)
  deal_ii_setup_target(bp3_restore_mesh)
  add_executable(test_bp3_environment test_execution_environment.cc)
  target_compile_features(test_bp3_environment PRIVATE cxx_std_17)
  add_executable(test_bp3_restored_model test_restored_model.cc)
  target_compile_features(test_bp3_restored_model PRIVATE cxx_std_17)
endif()
option(BP3_BUILD_RESEARCH_REFERENCE "Build the retained pre-cleanup research plugin" OFF)
if(BP3_BUILD_RESEARCH_REFERENCE)
  add_library(bp3_research SHARED reference_200km/bp3.cc)
  ASPECT_SETUP_PLUGIN(bp3_research)
endif()
option(BP3_BUILD_COUPLING_CHECKS "Build the existing focused surface/coupling test plugin" OFF)
if(BP3_BUILD_COUPLING_CHECKS)
  add_library(bp3_coupling_checks SHARED ../../../tests/phase_field_fault_surface_dynamic_pressure.cc)
  ASPECT_SETUP_PLUGIN(bp3_coupling_checks)
endif()
